/*
  4164 / 41256 DRAM Tester for Arduino UNO

  - ZIF socket: 16-pin 4164/41256 style
  - 2 LEDs: PASS/FAIL
  - Jumper selects mode:
      MODE_PIN (A2) tied to GND when jumper installed => 4164
      MODE_PIN floating (INPUT_PULLUP) => 41256

  Address mapping (fast port-mapped):
    A0..A5 -> D2..D7  (PORTD bits 2..7)
    A6..A8 -> D8..D10 (PORTB bits 0..2)  [A8 only used for 41256]

  Control:
    /RAS -> D11 (PORTB bit 3)
    /CAS -> D12 (PORTB bit 4)
    /WE  -> D13 (PORTB bit 5)

  Data:
    DIN  -> A0 (PORTC bit 0) output
    DOUT -> A1 (PORTC bit 1) input
*/

#include <Arduino.h>

// ----- User I/O -----
static const uint8_t MODE_PIN = A2;
static const uint8_t PASS_LED = A3;
static const uint8_t FAIL_LED = A4;

// ----- Port bit definitions (ATmega328P / UNO) -----
static const uint8_t DIN_BIT  = 0; // PC0 (A0)
static const uint8_t DOUT_BIT = 1; // PC1 (A1)

static const uint8_t RAS_BIT = 3;  // PB3 (D11)
static const uint8_t CAS_BIT = 4;  // PB4 (D12)
static const uint8_t WE_BIT  = 5;  // PB5 (D13)

// Small delays; UNO is slow anyway, but we keep sequencing clean.
static const uint8_t T_SETUP_US  = 1;
static const uint8_t T_PULSE_US  = 1;
static const uint8_t T_PRECHG_US = 1;

struct ChipCfg {
  const char* name;
  uint8_t rowBits;
  uint8_t colBits;
  uint8_t addrBits;     // total linear address bits (rowBits + colBits)
  bool useA8;           // drive A8 (D10) or tri-state it
  uint32_t size;        // number of bit addresses
};

static ChipCfg cfg4164  = { "4164",  8, 8, 16, false, (1UL << 16) };
static ChipCfg cfg41256 = { "41256", 9, 9, 18, true,  (1UL << 18) };

#define RAS_LOW()   (PORTB &= ~(1 << RAS_BIT))
#define RAS_HIGH()  (PORTB |=  (1 << RAS_BIT))
#define CAS_LOW()   (PORTB &= ~(1 << CAS_BIT))
#define CAS_HIGH()  (PORTB |=  (1 << CAS_BIT))
#define WE_LOW()    (PORTB &= ~(1 << WE_BIT))
#define WE_HIGH()   (PORTB |=  (1 << WE_BIT))

inline void dinWrite(uint8_t bitVal) {
  if (bitVal) PORTC |=  (1 << DIN_BIT);
  else        PORTC &= ~(1 << DIN_BIT);
}

inline uint8_t doutRead() {
  return (PINC & (1 << DOUT_BIT)) ? 1 : 0;
}

/*
  Present a multiplexed address value on A0..A7 (+ optional A8).
  - A0..A5 are on PORTD bits 2..7
  - A6..A8 are on PORTB bits 0..2
*/
inline void setAddrPins(uint16_t val, bool useA8) {
  // A0..A5 -> PD2..PD7
  uint8_t lo6 = (uint8_t)(val & 0x3F);
  PORTD = (PORTD & 0x03) | (lo6 << 2);

  // A6..A8 -> PB0..PB2
  uint8_t hi = (uint8_t)((val >> 6) & 0x07);
  if (!useA8) hi &= 0x03; // only A6..A7, keep A8 low (and no pull-up)
  PORTB = (PORTB & 0xF8) | hi; // preserve PB3..PB7 (control + oscillator)
}

/*
  One DRAM write cycle at linear address 'addr' writing single bit 'dataBit'.
  Uses early-write style: WE low before CAS.
*/
void dramWriteBit(const ChipCfg& c, uint32_t addr, uint8_t dataBit) {
  uint16_t rowMask = (1U << c.rowBits) - 1U;
  uint16_t colMask = (1U << c.colBits) - 1U;

  uint16_t col = (uint16_t)(addr & colMask);
  uint16_t row = (uint16_t)((addr >> c.colBits) & rowMask);

  // Prepare signals
  CAS_HIGH();
  RAS_HIGH();
  WE_LOW();            // select write mode
  dinWrite(dataBit);

  // Row
  setAddrPins(row, c.useA8);
  delayMicroseconds(T_SETUP_US);
  RAS_LOW();
  delayMicroseconds(T_PULSE_US);

  // Column
  setAddrPins(col, c.useA8);
  delayMicroseconds(T_SETUP_US);
  CAS_LOW();
  delayMicroseconds(T_PULSE_US);

  // End cycle
  CAS_HIGH();
  RAS_HIGH();
  WE_HIGH();
  delayMicroseconds(T_PRECHG_US);
}

/*
  One DRAM read cycle at linear address 'addr', returns bit.
*/
uint8_t dramReadBit(const ChipCfg& c, uint32_t addr) {
  uint16_t rowMask = (1U << c.rowBits) - 1U;
  uint16_t colMask = (1U << c.colBits) - 1U;

  uint16_t col = (uint16_t)(addr & colMask);
  uint16_t row = (uint16_t)((addr >> c.colBits) & rowMask);

  CAS_HIGH();
  RAS_HIGH();
  WE_HIGH(); // read mode

  // Row
  setAddrPins(row, c.useA8);
  delayMicroseconds(T_SETUP_US);
  RAS_LOW();
  delayMicroseconds(T_PULSE_US);

  // Column
  setAddrPins(col, c.useA8);
  delayMicroseconds(T_SETUP_US);
  CAS_LOW();
  delayMicroseconds(T_PULSE_US);

  uint8_t bitVal = doutRead();

  CAS_HIGH();
  RAS_HIGH();
  delayMicroseconds(T_PRECHG_US);
  return bitVal;
}

/*
  Quick sanity: write/read one address both 0 and 1.
*/
bool testSanity(const ChipCfg& c) {
  const uint32_t a = 0;

  dramWriteBit(c, a, 0);
  if (dramReadBit(c, a) != 0) return false;

  dramWriteBit(c, a, 1);
  if (dramReadBit(c, a) != 1) return false;

  return true;
}

/*
  Address-line test (classic):
  - Write 1 to base
  - Write 0 to each power-of-two offset
  - Verify base still 1 and offsets are 0
  - Invert and repeat
*/
bool testAddressLines(const ChipCfg& c) {
  const uint32_t base = 0;

  // Phase 1
  dramWriteBit(c, base, 1);
  for (uint8_t b = 0; b < c.addrBits; b++) {
    uint32_t a = (1UL << b);
    if (a >= c.size) break;
    dramWriteBit(c, a, 0);
  }

  if (dramReadBit(c, base) != 1) return false;
  for (uint8_t b = 0; b < c.addrBits; b++) {
    uint32_t a = (1UL << b);
    if (a >= c.size) break;
    if (dramReadBit(c, a) != 0) return false;
  }

  // Phase 2 (invert)
  dramWriteBit(c, base, 0);
  for (uint8_t b = 0; b < c.addrBits; b++) {
    uint32_t a = (1UL << b);
    if (a >= c.size) break;
    dramWriteBit(c, a, 1);
  }

  if (dramReadBit(c, base) != 0) return false;
  for (uint8_t b = 0; b < c.addrBits; b++) {
    uint32_t a = (1UL << b);
    if (a >= c.size) break;
    if (dramReadBit(c, a) != 1) return false;
  }

  return true;
}

/*
  Full cell test:
  For every address:
    write 0, read 0
    write 1, read 1
*/
bool testAllCells(const ChipCfg& c) {
  for (uint32_t a = 0; a < c.size; a++) {
    dramWriteBit(c, a, 0);
    if (dramReadBit(c, a) != 0) return false;

    dramWriteBit(c, a, 1);
    if (dramReadBit(c, a) != 1) return false;
  }
  return true;
}

void setResult(bool pass) {
  digitalWrite(PASS_LED, pass ? HIGH : LOW);
  digitalWrite(FAIL_LED, pass ? LOW  : HIGH);
}

void setupHardwareForChip(const ChipCfg& c) {
  // Configure address + control pins using DDR registers for speed
  // PORTD: D2..D7 outputs (A0..A5)
  DDRD |= 0b11111100;

  // PORTB:
  // PB0..PB1 always outputs for A6..A7
  // PB2 output only if using A8 (41256); otherwise input (tri-state)
  // PB3..PB5 outputs for /RAS /CAS /WE
  DDRB |= (1 << PB0) | (1 << PB1) | (1 << RAS_BIT) | (1 << CAS_BIT) | (1 << WE_BIT);
  if (c.useA8) DDRB |=  (1 << PB2);
  else        DDRB &= ~(1 << PB2);

  // Data: DIN output, DOUT input
  DDRC |=  (1 << DIN_BIT);
  DDRC &= ~(1 << DOUT_BIT);

  // Default inactive
  RAS_HIGH();
  CAS_HIGH();
  WE_HIGH();
  dinWrite(0);

  // LEDs/mode
  pinMode(MODE_PIN, INPUT_PULLUP);
  pinMode(PASS_LED, OUTPUT);
  pinMode(FAIL_LED, OUTPUT);
  setResult(false);
}

void setup() {
  Serial.begin(115200);
  pinMode(MODE_PIN, INPUT_PULLUP);
  pinMode(PASS_LED, OUTPUT);
  pinMode(FAIL_LED, OUTPUT);

  delay(250);
  Serial.println(F("\nDRAM Tester starting..."));
}

void loop() {
  // Read mode jumper
  bool mode41256 = (digitalRead(MODE_PIN) == HIGH); // jumper removed => HIGH
  const ChipCfg& c = mode41256 ? cfg41256 : cfg4164;

  setupHardwareForChip(c);

  Serial.print(F("Mode: "));
  Serial.println(c.name);

  // Run tests
  bool pass = true;

  Serial.println(F("Sanity test..."));
  pass &= testSanity(c);

  if (pass) {
    Serial.println(F("Address line test..."));
    pass &= testAddressLines(c);
  }

  if (pass) {
    Serial.println(F("Full cell test (this may take a few seconds)..."));
    pass &= testAllCells(c);
  }

  setResult(pass);

  Serial.print(F("RESULT: "));
  Serial.println(pass ? F("PASS") : F("FAIL"));

  // Wait before re-running (and allow chip swap)
  delay(1500);
}
